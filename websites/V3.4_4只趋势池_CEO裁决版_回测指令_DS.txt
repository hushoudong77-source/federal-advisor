### 判定规则（每月最后一个交易日计算，下月生效）

```python
def get_regime(date):
    # 取SPX和沪深300的月频MA120乖离率
    spx_bias = (SPX_CLOSE / SPX_MA120 - 1)
    hs300_bias = (HS300_CLOSE / HS300_MA120 - 1)
    
    # 取两者中更悲观的方向
    combined_bias = min(spx_bias, hs300_bias)
    
    if combined_bias > 0.05:       return 'BULL'    # 牛市
    elif combined_bias < -0.05:    return 'BEAR'    # 熊市
    else:                          return 'RANGE'   # 震荡
```

## 三、入场条件（统一公式）

```python
def check_entry(ticker, adx, close, ma20, bias, regime):
    # 1. 基本方向
    if close <= ma20:
        return False
    
    # 2. 乖离率过滤（按regime动态调整上限）
    if regime == 'BEAR':
        bias_max = 0.08
        bias_min = -0.03
    else:
        bias_max = 0.10
        bias_min = -0.02
    
    if bias < bias_min or bias > bias_max:
        return False
    
    # 3. ADX门槛（按标的+regime查表）
    adx_thresholds = {
        'QQQ':    {'BULL':18,'RANGE':20,'BEAR':22},
        'IVV':    {'BULL':18,'RANGE':20,'BEAR':22},
        'EWY':    {'BULL':22,'RANGE':22,'BEAR':25},
        '518880': {'BULL':18,'RANGE':20,'BEAR':22},
    }
    threshold = adx_thresholds[ticker][regime]
    
    if adx < threshold:
        return False
    
    # 4. 冷却期检查
    if weeks_since_last_close(ticker) < 3:
        return False
    
    # 5. 仓位上限检查
    if count_active_l2() >= max_positions[regime]:
        return False
    
    if ticker in ['QQQ','IVV','EWY'] and count_overseas() >= overseas_limit[regime]:
        return False
    
    return True
```

def check_exit(ticker, close, ma20, entry_price, atr, regime):
    # P1: 绝对止损 -4%
    if close <= entry_price * 0.96:
        return ('STOP', '单标止损-4%')
    
    # P2: 周频组合止损 -6%（全L2组合当周累计）
    if weekly_pnl_L2 <= -0.06:
        return ('COMBO_STOP', '组合周止损-6%，全部L2清仓')
    
    # P3: 跌破MA20（核心出场信号）
    if close < ma20:
        return ('MA20_BREAK', '跌破MA20')
    
    # P4: 乖离率过热（牛市>15% / 震荡>12% / 熊市>10%）

    overheat = {'BULL':0.15, 'RANGE':0.12, 'BEAR':0.10}
    if bias > overheat[regime]:
        return ('OVERHEAT', f'乖离率过热 {bias:.1%}')
    
    return (None, 'HOLD')
