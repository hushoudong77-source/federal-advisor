# ─── 实验组A：启用 Regime Switching ───

def get_regime_A(date):
    """每月最后一个交易日判定，当月有效"""
    spx_ma120 = MA(SPX_CLOSE, 120)
    spx_bias  = (SPX_CLOSE[-1] - spx_ma120) / spx_ma120
    
    csi300_ma120 = MA(CSI300_CLOSE, 120)
    csi300_bias  = (CSI300_CLOSE[-1] - csi300_ma120) / csi300_ma120
    
    # 取两者中更差的环境
    min_bias = min(spx_bias, csi300_bias)
    
    if min_bias > 0.05:
        return 'BULL'
    elif min_bias < -0.05:
        return 'BEAR'
    else:
        return 'SIDEWAYS'

def get_params_A(regime):

# ─── 共用入场条件 ───

def check_entry(ticker, close, ma20, adx, bias, params):
    # 1. 价格必须在MA20上方
    if close <= ma20:
        return False
    
    # 2. 乖离率在[-2%, 10%]区间
    if bias < -0.02 or bias > params['bias_upper']:
        return False
    
    # 3. ADX门槛（按标的类型）
    if ticker == 'EWY':
        if adx < params['adx_ewy']:
            return False
    else:
        if adx < params['adx_main']:
            return False
    
    return True

# ─── 共用出场条件 ───

def check_exit(pos, close, ma20, current_week_pnl):
    # 1. 单标止损
    if pos.unrealized_pnl_pct <= STOP_LOSS_SINGLE:
        return True, 'STOP_LOSS'
    
    # 2. 跌破MA20
    if close < ma20:
        return True, 'BELOW_MA20'
    
    # 3. 组合周止损（所有L2持仓合计）
    if current_week_pnl <= STOP_LOSS_WEEKLY:
        return True, 'PORTFOLIO_STOP'
    
    return False, None

# ─── L2主循环（两组共用交易逻辑，仅参数不同）───

def run_L2(date, regime_params, label):
    """
    label: 'A' 或 'B'
    regime_params: get_params_A(regime) 或 get_params_B()
    """
    CURRENT_EQUITY = CASH + sum(pos.shares * price(pos.ticker) for pos in L2_POS)
    L2_BUDGET = CURRENT_EQUITY * L2_WEIGHT
    
    # Step 1: 检查出场
    for pos in L2_POS[:]:
        close = get_close(pos.ticker, date)
        ma20  = MA(get_closes(pos.ticker, 20))
        week_pnl = sum(p.unrealized_pnl_pct * p.weight for p in L2_POS)
        
        should_exit, reason = check_exit(pos, close, ma20, week_pnl)
        if should_exit:
            close_position(pos, reason)
            pos.cooldown_until = date + COOLDOWN_WEEKS
    
    # Step 2: 空仓生息
    if len(L2_POS) == 0:
        CASH *= (1 + 0.025 / 52)  # 511260 国债ETF周化
    
    # Step 3: 扫描入场
    if len(L2_POS) >= regime_params['max_fire']:
        return  # 已达开火上限
    
    candidates = []
    for ticker, cfg in L2_POOL.items():
        if ticker in [p.ticker for p in L2_POS]:
            continue  # 已持仓
        if ticker in cooldown_map and date < cooldown_map[ticker]:
            continue  # 冷却中
        
        closes = get_closes(ticker, 120)
        close = closes[-1]
        ma20  = MA(closes, 20)
        adx   = calc_ADX(closes, 14)
        bias  = (close - ma20) / ma20
        
        if check_entry(ticker, close, ma20, adx, bias, regime_params):
            candidates.append({
                'ticker': ticker,
                'adx': adx,
                'bias': bias,
                'type': cfg['type'],
            })
    
    # 排序：ADX降序
    candidates.sort(key=lambda x: x['adx'], reverse=True)
    
    # 海外名额限制
    overseas_selected = 0
    selected = []
    for c in candidates:
        if c['type'] == 'US':
            if overseas_selected >= regime_params['max_overseas']:
                continue
            overseas_selected += 1
        selected.append(c)
        if len(selected) >= regime_params['max_fire'] - len(L2_POS):
            break
    
    # 开仓
    for c in selected:
        weight = L2_POOL[c['ticker']]['max_weight']
        target_value = CURRENT_EQUITY * weight
        shares = int(target_value / c['close'])
        open_position(c['ticker'], shares, c['close'], date)
