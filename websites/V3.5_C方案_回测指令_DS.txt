```python
def generate_signals_A(friday_date):
    candidates = []
    for ticker in [QQQ, EWY, IVV, 518880]:
        adx = calc_adx(ticker, period=14, end=friday_date)  # 日频ADX
        ma20 = calc_sma(ticker, period=20, end=friday_date)  # 日频MA20
        close = get_close(ticker, friday_date)
        bias = close / ma20 - 1

        if close <= ma20: continue
        if bias < -0.02 or bias > 0.10: continue

        if ticker == EWY:
            if adx < 22: continue
        else:
            if adx < 18: continue

        if ticker in [QQQ, IVV]:
            proxy = get_proxy_index(ticker)
            breadth = calc_breadth(proxy, friday_date)
            if breadth is None: breadth = 1.0
            if breadth < 0.35 and adx < 25: continue

        candidates.append({ticker: ticker, adx: adx, bias: bias})

    candidates.sort(key=lambda x: x[adx], reverse=True)

    overseas = []
    domestic = []
    for c in candidates:
        if c[ticker] in [QQQ, EWY, IVV]:
            if len(overseas) < 3:
                overseas.append(c)
        else:
            domestic.append(c)
        if len(overseas) + len(domestic) >= 3:
            break

    return overseas + domestic
```

4.1 周线阶段判定（纯物理指标，0缠论）

```python
def calc_weekly_stage(ticker, friday_date):
    """
    基于周线数据判定标的当前阶段。
    每周五收盘后，读取该标的最新的周线K线数据计算。
    周线ADX(14)、周线MA20、周线MA60、周线收盘价。
    """
    weekly_close = get_weekly_close(ticker, friday_date)
    weekly_ma20 = calc_weekly_sma(ticker, period=20, end=friday_date)
    weekly_ma60 = calc_weekly_sma(ticker, period=60, end=friday_date)
    weekly_adx = calc_weekly_adx(ticker, period=14, end=friday_date)
    weekly_bias = weekly_close / weekly_ma20 - 1

    # 进攻态：周线趋势确认，日频信号可信
    if weekly_adx > 25 and weekly_close > weekly_ma20 and weekly_bias > 0.05:
        return 进攻态

    # 危险态：周线趋势衰竭或破位，禁止入场，已持仓强制平仓
    if weekly_adx < 15 or weekly_close < weekly_ma60 or weekly_bias < -0.10:
        return 危险态

    # 防御态：周线震荡，不新增，已持仓可继续持有
    # 条件：ADX在[15,25]之间，乖离率在[-5%,5%]之间
    if 15 <= weekly_adx <= 25 and -0.05 <= weekly_bias <= 0.05:
        return 防御态

    # 其他模糊状态，默认防御态（不新增）
    return 防御态
```

```python
def generate_signals_B(friday_date):
    candidates = []

    for ticker in [QQQ, EWY, IVV, 518880]:
        # ========== 第一层：周线预筛 ==========
        stage = calc_weekly_stage(ticker, friday_date)

        if stage == 危险态:
            # 危险态：直接跳过，不参与日频排序
            # 若已持仓，在execute阶段强制平仓
            continue

        if stage == 防御态:
            # 防御态：不新增，但已持仓可继续持有
            # 不加入candidates，但不清仓（在execute中处理）
            continue

        # 进攻态：进入第二层日频过滤
        # ========== 第二层：日频竞争（与A组完全一致） ==========
        adx = calc_adx(ticker, period=14, end=friday_date)
        ma20 = calc_sma(ticker, period=20, end=friday_date)
        close = get_close(ticker, friday_date)
        bias = close / ma20 - 1

        if close <= ma20: continue
        if bias < -0.02 or bias > 0.10: continue

        if ticker == EWY:
            if adx < 22: continue
        else:
            if adx < 18: continue

        if ticker in [QQQ, IVV]:
            proxy = get_proxy_index(ticker)
            breadth = calc_breadth(proxy, friday_date)
            if breadth is None: breadth = 1.0
            if breadth < 0.35 and adx < 25: continue

        candidates.append({ticker: ticker, adx: adx, bias: bias, stage: stage})

    # ADX排序
    candidates.sort(key=lambda x: x[adx], reverse=True)

    # 名额分配（与A组一致）
    overseas = []
    domestic = []
    for c in candidates:
        if c[ticker] in [QQQ, EWY, IVV]:
            if len(overseas) < 3:
                overseas.append(c)
        else:
            domestic.append(c)
        if len(overseas) + len(domestic) >= 3:
            break

    return overseas + domestic
```

```python
def execute_open(monday_date, selected, portfolio):
    CURRENT_EQUITY = calc_equity()
    L2_LIMIT = CURRENT_EQUITY * 0.25
    max_fire = 3

    for s in selected:
        if s[ticker] not in [p[ticker] for p in portfolio]:
            buy_price = get_open(s[ticker], monday_date)
            buy_price *= (1 + friction_rate(s[ticker]) / 2)

            slot_cap = L2_LIMIT / max_fire
            max_single = 0.05 if s[ticker] == QQQ else 0.03
            max_cap = CURRENT_EQUITY * max_single

            position_size = min(slot_cap, max_cap)
            shares = int(position_size / buy_price)

            total_cost = shares * buy_price
            if CASH >= total_cost and shares > 0:
                open_position(s[ticker], buy_price, shares)
                CASH -= total_cost
```

        if calc_weekly_stage(ticker, friday_date) == 危险态:
            sell_price = get_open(ticker, monday_date) * (1 - friction/2)
            close_position(pos, sell_price, reason=WEEKLY_DANGER)
            CASH += pos[shares] * sell_price
            continue

        # 排名淘汰
        if ticker not in [s[ticker] for s in selected]:
            sell_price = get_open(ticker, monday_date) * (1 - friction/2)
            close_position(pos, sell_price)
            CASH += pos[shares] * sell_price
            continue

        # -4%止损
        current = get_open(ticker, monday_date)
        if current / pos[entry_price] - 1 <= -0.04:
            sell_price = current * (1 - friction/2)
            close_position(pos, sell_price, reason=STOP)
            CASH += pos[shares] * sell_price

补丁6：动态权益

def calc_equity(cash, positions, prices):
    equity = cash
    for pos in positions:
        equity += pos[shares] * prices.get(pos[ticker], 0)
    return equity

CASH_YIELD_RATE = 0.025

def apply_cash_yield(cash, l2_positions, l3_active):
    if len(l2_positions) == 0 and not l3_active:
        return cash * (1 + CASH_YIELD_RATE / 52)
    return cash
